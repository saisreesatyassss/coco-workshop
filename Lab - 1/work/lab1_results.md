Naive SQL:
```sql
SELECT
    C_MKTSEGMENT,
    SUM(O_TOTALPRICE) AS TOTAL_REVENUE
FROM
    SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.CUSTOMER
JOIN
    SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.ORDERS
ON
    C_CUSTKEY = O_CUSTKEY
GROUP BY
    C_MKTSEGMENT
```
```sql
SELECT
    C.C_MKTSEGMENT,
    SUM(L.L_EXTENDEDPRICE * (1 - L.L_DISCOUNT)) AS COST_OF_GOODS_SOLD
FROM
    SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.CUSTOMER C
JOIN
    SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.ORDERS O ON C.C_CUSTKEY = O.O_CUSTKEY
JOIN
    SNOWFLAKE_SAMPLE_DATA.TPCH_SF1.LINEITEM L ON O.O_ORDERKEY = L.L_ORDERKEY
GROUP BY
    C.C_MKTSEGMENT
```
